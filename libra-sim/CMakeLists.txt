cmake_minimum_required(VERSION 3.20)
project(libra_sim LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

option(LIBRA_BUILD_PYTHON "Build the pybind11 module" ON)
option(LIBRA_BUILD_TESTS "Build the C++ tests" ON)

add_library(libra_sim STATIC src/types.cpp src/bitboard.cpp src/position.cpp)
target_include_directories(libra_sim PUBLIC include)
set_target_properties(libra_sim PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(LIBRA_BUILD_TESTS)
  enable_testing()
  add_executable(test_rules tests/test_rules.cpp)
  target_link_libraries(test_rules PRIVATE libra_sim)
  add_test(NAME rules COMMAND test_rules)
  add_executable(perft tests/perft.cpp)
  target_link_libraries(perft PRIVATE libra_sim)
  add_test(NAME perft COMMAND perft --check)
endif()

if(LIBRA_BUILD_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_sim python/bindings.cpp)
  target_link_libraries(_sim PRIVATE libra_sim)
  if(SKBUILD)
    install(TARGETS _sim DESTINATION librashogi)
  else()
    set_target_properties(_sim PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/python/librashogi)
  endif()
endif()
