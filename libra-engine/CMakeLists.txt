cmake_minimum_required(VERSION 3.20)
project(libra_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TARGET libra_search)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../libra-search ${CMAKE_BINARY_DIR}/libra-search)
endif()
set(LIBRA_ORT_DIR "" CACHE PATH "ONNX Runtime の展開ディレクトリ（include/ と lib/ を持つ。tools/fetch_onnxruntime.sh）")
if(NOT LIBRA_ORT_DIR OR NOT EXISTS "${LIBRA_ORT_DIR}/include/onnxruntime_c_api.h")
  message(FATAL_ERROR "LIBRA_ORT_DIR に ONNX Runtime の展開先を指定してください（tools/fetch_onnxruntime.sh）")
endif()
add_executable(libra src/main.cpp src/engine.cpp src/ort_infer.cpp)
target_include_directories(libra PRIVATE ${LIBRA_ORT_DIR}/include src)
target_link_libraries(libra PRIVATE libra_search ${CMAKE_DL_LIBS})
target_compile_options(libra PRIVATE -Wall -Wextra)
if(MINGW)
  # 依存ライブラリは静的リンクし、ONNX Runtime（と DirectML）の DLL だけを同梱する（docs/libra-local.md §6.2、decisions.md 2026-09-14）
  target_link_options(libra PRIVATE -static -static-libgcc -static-libstdc++)
endif()
# 実行時にロードする ONNX Runtime の共有ライブラリを実行ファイルの隣へ写す
if(WIN32)
  file(GLOB ORT_RUNTIME "${LIBRA_ORT_DIR}/lib/*.dll")
else()
  file(GLOB ORT_RUNTIME "${LIBRA_ORT_DIR}/lib/*.so*")
endif()
foreach(f ${ORT_RUNTIME})
  add_custom_command(TARGET libra POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${f}" "$<TARGET_FILE_DIR:libra>/")
endforeach()
