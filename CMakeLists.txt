cmake_minimum_required(VERSION 3.20)
project(LibraShogi LANGUAGES CXX)
option(LIBRA_BUILD_PYTHON "Build the pybind11 modules" ON)
option(LIBRA_BUILD_TESTS "Build the C++ tests" ON)
option(LIBRA_BUILD_ENGINE "Build the USI engine libra/libra.exe (needs LIBRA_ORT_DIR)" ON)
if(LIBRA_BUILD_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
endif()
add_subdirectory(libra-sim)
add_subdirectory(libra-search)
if(LIBRA_BUILD_ENGINE AND LIBRA_ORT_DIR)
  add_subdirectory(libra-engine)
else()
  message(STATUS "libra-engine: skipped (set -DLIBRA_ORT_DIR=<onnxruntime dir> to build)")
endif()
