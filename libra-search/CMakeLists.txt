cmake_minimum_required(VERSION 3.20)
project(libra_search LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TARGET libra_sim)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../libra-sim ${CMAKE_BINARY_DIR}/libra-sim)
endif()
find_package(Threads REQUIRED)
add_library(libra_search STATIC src/selfplay.cpp src/dfpn.cpp)
target_include_directories(libra_search PUBLIC include)
target_link_libraries(libra_search PUBLIC libra_sim Threads::Threads)
set_target_properties(libra_search PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
  target_compile_options(libra_search PRIVATE -Wall -Wextra)
endif()

if(LIBRA_BUILD_TESTS)
  enable_testing()
  add_executable(test_dfpn tests/test_dfpn.cpp)
  target_link_libraries(test_dfpn PRIVATE libra_search)
  add_test(NAME dfpn COMMAND test_dfpn)
endif()

if(LIBRA_BUILD_PYTHON)
  pybind11_add_module(_search python/bindings.cpp)
  target_link_libraries(_search PRIVATE libra_search)
  if(SKBUILD)
    install(TARGETS _search DESTINATION librasearch)
  else()
    set_target_properties(_search PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/python/librasearch)
  endif()
endif()
